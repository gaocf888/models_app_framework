WITH over_temp_level AS (

-- 替换为你的超温事件时间范围
t.start_time >= '2026-01-01 00:00:00'
AND t.highest_temp > t.limit_temp
)
SELECT
over_level,
GROUP_CONCAT(
CONCAT(pi_code, '（位置：', boiler_name, '-',  point_name, '，第', row_num, '排第', pipe_num, '根）'),
'、'
) AS 测点及位置列表,
COUNT(DISTINCT pi_code) AS 测点数量
FROM over_temp_level
WHERE over_level != '正常'

WHEN '中度超温' THEN 2

四、锅炉超温核心原因智能诊断（大模型推理·多测点分类归因）
说明：按“共性原因（所有超温测点共同诱因）+ 区域专属原因（特定区域测点超温诱因）”分类，避免单一归因，贴合多测点场景
（一）共性原因（多测点超温共同诱因）
烟气侧：炉膛整体火焰中心上移，上部受热面整体烟温偏高；炉膛整体积灰、结焦，换热效率下降，导致多区域受热面超温
介质侧：整体减温水系统调节滞后，或总减温水投入量不足，无法满足全炉膛控温需求；机组负荷波动过大，介质流量整体分配不均
运行操作：煤质波动（高挥发分、高灰分），导致整体燃烧工况异常；风量配比不合理，炉膛氧量控制偏差，影响整体换热
设备本体：多区域受热面管材老化，耐热性能整体下降；测温元件批量漂移（若多测点误报，需单独标注）
（二）区域专属原因（按超温区域分类）
区域1（如：右侧末级过热器）：
烟气侧：局部火焰偏斜，右侧烟温偏高，冲刷该区域受热面；局部结焦严重，换热变差
介质侧：该区域减温水调门开度不足、调节失效，或管内结垢、氧化皮堆积，通流截面减小
设备侧：该区域导流装置异常，介质流量分配偏小
区域2（如：前墙水冷壁）：
烟气侧：前墙火焰冲刷剧烈，局部烟温过高；积灰未及时清理，换热受阻
运行侧：该区域吹灰频次不足，积灰累积；启停磨煤机时，局部燃料堆积，燃烧不均
区域3（如：低温再热器）：（按需补充，对应具体超温区域）
五、超温带来的安全危害评估（多测点叠加影响）
短期危害：多区域受热面金属蠕变加速，氧化皮大量脱落，可能堵塞局部管道，加剧局部超温，形成“超温-堵塞-更严重超温”恶性循环
中期危害：不同区域管材金相组织劣化、疲劳损伤累积，检修范围扩大，检修周期缩短，增加检修成本
长期危害：多个超温测点对应区域管壁易出现鼓包、裂纹，若未及时处置，可能引发多区域受热面同时爆管，导致机组非计划停机，损失严重
经济影响：非计划停机损失、抢修成本、发电量损失、能耗上升（多区域超温导致换热效率下降，能耗增幅高于单一测点超温）
环保影响：超温工况下燃烧不完全，NOx、粉尘排放波动超标，且多区域超温易导致燃烧工况不稳定，排放超标持续时间更长
六、大模型智能处置调控措施（分优先级·多测点差异化处置）
核心原则：先处置严重超温测点，再优化中度/轻微超温测点；兼顾共性问题整改，同步解决区域专属问题，避免顾此失彼
（一）紧急处置（严重超温测点立即执行）
整体调控：适当降低机组负荷，放缓升温速率，避免整体工况进一步扰动，缓解所有超温测点压力
差异化降温：对严重超温测点对应区域，开大该回路减温水，精准控温；若多区域严重超温，优先保障关键受热面（如末级过热器、再热器）减温水供应
紧急调整：调整炉膛配风，压低火焰中心，消除局部烟气偏流，缓解集中超温区域的烟温压力
（二）运行优化调整（兼顾所有超温测点）
共性优化：优化煤种掺烧，稳定入炉煤热值；调整炉膛氧量、二次风配比，消除整体烟气偏流；平稳升降负荷，杜绝大幅参数扰动
区域优化：
对集中超温区域，增加吹灰频次，重点清理结焦、积灰；对零散超温测点，排查局部吹灰盲区
针对不同区域专属问题，调整对应回路减温水调节参数，优化介质流量分配；对管内可能结垢的区域，加强水质监测
（三）检修预防措施（分区域落实）
重点检修：停机后对严重超温区域管壁进行测厚、金相检测，排查裂纹、鼓包隐患；清理管内氧化皮、管道疏通
全面排查：对所有超温测点对应区域的受热面、焊口、导流装置进行全面检查，更换老化管材、失效测温元件
测点校验：批量校验所有壁温测点，消除测点失真、误报/漏报问题，确保多测点数据准确性
（四）长效防控方案（适配多测点监测）
大模型优化：增设多测点联动预警阈值，对同一区域多个测点同时超温、不同区域零散超温分别设置预警逻辑，提前预判超温
分区管控：建立机组不同负荷下，各区域受热面最优温度工况模型，实现分区域精准控温
制度完善：制定分区域受热面防超温标准操作卡，明确不同超温场景（集中/分散）的处置流程，规范运行操作
七、整改完成情况&效果验证（多测点全覆盖）
已执行调控操作：（分区域列明）如“右侧末级过热器减温水开度调整至55%、前墙水冷壁补充吹灰1次、整体负荷下调至450MW”等
全测点效果验证：
严重超温测点：____个已降至允许范围（当前温度____℃），剩余____个（若有）的降温进度
中度/轻微超温测点：全部降至允许范围，当前平均壁温____℃
关联参数验证：烟温、汽温、减温水参数恢复稳定性，各区域烟气偏差、介质流量分配趋于合理
后续跟踪监测时长：____小时，重点跟踪原超温测点及同区域关联测点，防止复温
WITH temp_after AS (
SELECT
t.pi_code,
t.limit_temp,
t.highest_temp,-- 判定超温等级
CASE
WHEN ( t.highest_temp - t.limit_temp ) >= 20 THEN
'严重超温'
WHEN ( t.highest_temp - t.limit_temp ) >= 10
AND ( t.highest_temp - t.limit_temp ) < 20 THEN '中度超温' WHEN ( t.highest_temp - t.limit_temp ) >= 5
AND ( t.highest_temp - t.limit_temp ) < 10 THEN
'轻微超温' ELSE '正常'
END AS over_level
FROM
monitor_hotarea_temp t -- 关联锅炉表，用名称过滤
JOIN account_boiler ab ON t.boiler_id = ab.boiler_id
WHERE-- 【这里只改机组名称就行，例如 #1机组、#2机组】
ab.boiler_name = '#1号锅炉' -- 替换为你调控后的时间
AND t.start_time >= '2026-01-01 12:00:00'
) SELECT-- 严重超温恢复情况
COUNT( CASE WHEN over_level = '严重超温' AND highest_temp <= limit_temp THEN 1 END ) AS 已恢复严重超温数,
MAX( CASE WHEN over_level = '严重超温' AND highest_temp <= limit_temp THEN highest_temp END ) AS 恢复后最高壁温,
COUNT( CASE WHEN over_level = '严重超温' AND highest_temp > limit_temp THEN 1 END ) AS 剩余未恢复严重超温数,-- 中/轻度超温恢复情况
CASE
WHEN COUNT( CASE WHEN over_level IN ( '中度超温', '轻微超温' ) AND highest_temp > limit_temp THEN 1 END ) = 0 THEN
'是' ELSE '否'
END AS 中轻度是否全部恢复,
ROUND( AVG( CASE WHEN over_level IN ( '中度超温', '轻微超温' ) THEN highest_temp END ), 1 ) AS 中轻度当前平均壁温
FROM
temp_after;

多测点温度对照表（含设计值、实测值、超温差值、持续时长）
SELECT
t.pi_code AS 测点编号,
CONCAT(ab.boiler_name,'-',asd.device_name,'-',btp.point_name) AS 测点位置,
btd.over_hot_limit AS 设计壁温_℃,
MAX(t.highest_temp) AS 实测最高壁温_℃,
MAX(t.highest_temp) - btd.over_hot_limit AS 超温差值_℃,
SUM(t.limit_duration) AS 累计持续时长_秒
FROM monitor_hotarea_temp t
LEFT JOIN account_boiler ab ON t.boiler_id = ab.boiler_id
LEFT JOIN account_static_device asd ON t.device_id = asd.device_id
LEFT JOIN base_temp_device btd ON t.device_id = btd.device_id
LEFT JOIN base_temp_point btp ON t.pi_code = btp.point_code
WHERE
t.start_time BETWEEN '2026-01-01 00:00:00' AND '2026-06-02 00:00:00'
AND t.highest_temp > t.limit_temp
GROUP BY t.pi_code,ab.boiler_name,asd.device_name,btp.point_name,btd.over_hot_limit
ORDER BY 超温差值_℃ DESC;
